    const next={...all[idx],rememberHash:hash,rememberAt:Date.now()};

  }catch(err){ console.warn('No se pudo activar Recordarme',err); return false; }

function v89ForgetRemember(id){

  }catch{}
}
async function v89TryRememberedLogin(){
  try{
    const id=localStorage.getItem(V89_REMEMBER_ID);
    const token=localStorage.getItem(V89_REMEMBER_TOKEN);
    if(!id||!token)return false;
    const a=accountById(id);
    if(!a||!a.rememberHash||!a.rememberAt){v89ForgetRemember(id);return false}
    if(Date.now()-Number(a.rememberAt)>V89_REMEMBER_MAX_AGE){v89ForgetRemember(id);return false}
    const hash=await v89HashToken(token);
    if(hash!==a.rememberHash){v89ForgetRemember(id);return false}
    if(!a.cloudKey || a.cloudKeyVersion!==V131_ACCOUNT_PROFILE_VERSION){

      // cloud antigua, pedimos el PIN una vez para migrarlo de forma segura a V131.
      v89ForgetRemember(id);

    await enterAccount(a,null,false,true);
    showToast('🔓 Sesión recordada','La cuenta se ha abierto automáticamente en este dispositivo.');

async function enterAccount(account,pin,remember=false,fromRemember=false){
  activeAccountId=account.id;
  window.__marvelPrimaryAccount=(norm(account.person1)==='juanjo' && norm(account.person2)==='maria');
  LS=makeLS(account.id);
  localStorage.setItem(ACTIVE_ACCOUNT_STORE,account.id);
  if(pin!=null){
    const previousCloudKey=account.cloudKey||'';
    const previousLegacyKey=account.cloudKeyLegacy||'';
    const previousLegacyKey2=account.cloudKeyLegacy2||'';
    const cloudKey=await v90AccountCloudKey(account,pin);
    const stored=accountById(account.id)||account;
    window.__marvelAccountCloudKey=cloudKey;
    window.__marvelLegacyAccountCloudKey=(stored.cloudKeyLegacy && stored.cloudKeyLegacy!==cloudKey)?stored.cloudKeyLegacy:((previousCloudKey && previousCloudKey!==cloudKey)?previousCloudKey:'');
    window.__marvelLegacyAccountCloudKey2=(previousLegacyKey2 && previousLegacyKey2!==cloudKey && previousLegacyKey2!==window.__marvelLegacyAccountCloudKey)?previousLegacyKey2:(previousLegacyKey && previousLegacyKey!==cloudKey && previousLegacyKey!==window.__marvelLegacyAccountCloudKey?previousLegacyKey:'');
    if(remember)await v89RememberAccount({...account,cloudKey},cloudKey);
    else v89ForgetRemember(account.id);
  }else if(fromRemember){
    window.__marvelAccountCloudKey=account.cloudKey||'';
    window.__marvelLegacyAccountCloudKey=account.cloudKeyLegacy||'';
    window.__marvelLegacyAccountCloudKey2=account.cloudKeyLegacy2||'';
  }else{
    window.__marvelAccountCloudKey=account.cloudKey||'';
    window.__marvelLegacyAccountCloudKey=account.cloudKeyLegacy||'';
    window.__marvelLegacyAccountCloudKey2=account.cloudKeyLegacy2||'';
    v89ForgetRemember(account.id);
  }
  /*
   * V1.8: el PIN solamente autentica la cuenta. La pantalla de sincronización
   * NO puede envolver a init(). En versiones anteriores init() podía quedarse
   * ocupado en renderCatalog/heavy y, aunque la nube tuviera timeout, nunca
   * llegaba a ejecutar el código que quitaba #pinGate.
   *
   * Liberamos primero la interfaz y después inicializamos el contenido en una
   * tarea independiente. De esta forma el acceso a la página de inicio es una
   * garantía incluso si cualquier capa de datos se queda colgada.
   */

  document.body.classList.remove('locked');
  window.scrollTo({top:0,behavior:'auto'});

  setTimeout(async()=>{
    try{
      await init();
    }catch(error){
      // El arranque de una cuenta nunca debe dejar la aplicación bloqueada.
      // Si una capa local o cloud falla, la interfaz ya está desbloqueada.
      console.error('MARVEL account init · fallback local',error);
      try{
        if(!app.state?.items) app.state=clone(INITIAL_STATE);
        if(!app.checkpoint) app.checkpoint=clone(app.state);
        loadDerived();
        applySettings();
        try{renderCatalog()}catch(renderError){console.warn('MARVEL fallback renderCatalog',renderError)}
        try{heavy()}catch(heavyError){console.warn('MARVEL fallback heavy',heavyError)}
      }catch(fallbackError){
        console.error('MARVEL emergency local fallback',fallbackError);
      }
      try{cloudStatus('⚠️ Nube no disponible · usando copia local','err')}catch{}
      // V1.10.13: no mostramos un falso error de sincronización aquí.
      // Este catch cubre también errores de render/local init; la sincronización
      // cloud se reintenta de forma independiente por el coordinador final.
    }
  },0);
}

function seedProfileDefaultsForAccount(account, settings){
 const s=settings||{};
 const p1=String(account?.person1||'').trim()||'Persona 1';
 const p2=String(account?.person2||'').trim()||'Persona 2';
 const placeholder1=v=>!String(v??'').trim()||['persona 1','persona1'].includes(norm(v));
 const placeholder2=v=>!String(v??'').trim()||['persona 2','persona2'].includes(norm(v));
 let changed=false;
 if(placeholder1(s.person1) && p1!=='Persona 1'){s.person1=p1;changed=true;}
 if(placeholder2(s.person2) && p2!=='Persona 2'){s.person2=p2;changed=true;}
 if(typeof s.relation!=='string'||!s.relation.trim()){s.relation='pareja';changed=true;}
 // La ventana automática nace desactivada en todos los perfiles nuevos o
 // antiguos que todavía no tengan una preferencia guardada. Después el usuario
 // puede cambiarla desde Ajustes → Personalización.
 if(s.quickRating==null||s.quickRating===''){
   s.quickRating='off';
   changed=true;
 }
 // Vacío = automático. No congelamos aquí las iniciales para que siempre
 // evolucionen con el nombre hasta que el usuario introduzca una abreviatura.
 if(typeof s.initial1!=='string'){s.initial1='';changed=true;}
 if(typeof s.initial2!=='string'){s.initial2='';changed=true;}
 return changed;
}
